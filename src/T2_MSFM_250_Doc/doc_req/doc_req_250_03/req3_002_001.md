아래는 최종 확정된 공통 함수 전체 구현입니다.
기존의 cfgSet, T2_20_Cfg_StrCopy, parseDspConfig, serializeDspConfig, 밴드 처리 코드 등을 J2V / V2J 네이밍으로 완전히 대체합니다.

```cpp
// ============================================================================
// 공통 JSON ↔ C++ 변환 유틸리티 (J2V / V2J)
// 접두사: T2_20_Cfg_
// ============================================================================

#include <ArduinoJson.h>
#include <cstring>
#include <bitset>

// ----------------------------------------------------------------------------
// 1. 문자열
// ----------------------------------------------------------------------------

// J2V : JSON 필드 → C 문자열(char 배열) 복사
static inline void T2_20_Cfg_J2V_Str(JsonVariantConst p_v, char* p_dest, size_t p_size) {
    if (!p_v.isNull()) {
        const char* v_src = p_v.as<const char*>();
        if (v_src) strlcpy(p_dest, v_src, p_size);
    }
}

// V2J : C 문자열 → JSON 객체의 지정 키에 할당
static inline void T2_20_Cfg_V2J_Str(JsonObject p_obj, const char* p_key, const char* p_src) {
    p_obj[p_key] = p_src;
}

// ----------------------------------------------------------------------------
// 2. 일반 값 (스칼라, enum, bitset)
// ----------------------------------------------------------------------------

// J2V : JSON variant → 일반 변수 (null이면 무시)
template<typename T>
static inline void T2_20_Cfg_J2V_Val(JsonVariantConst p_v, T& p_dest) {
    if (!p_v.isNull()) p_dest = p_v.as<T>();
}

// bitset::reference 특수화
static inline void T2_20_Cfg_J2V_Val(JsonVariantConst p_v, std::bitset<16>::reference p_dest) {
    if (!p_v.isNull()) p_dest = p_v.as<bool>();
}

// V2J : 일반 값 → JSON 객체의 지정 키에 할당
template<typename T>
static inline void T2_20_Cfg_V2J_Val(JsonObject p_obj, const char* p_key, const T& p_val) {
    p_obj[p_key] = p_val;
}

// ----------------------------------------------------------------------------
// 3. 1차원 고정 크기 배열
// ----------------------------------------------------------------------------

// J2V : JSON 배열(또는 단일 값) → C 배열
template<typename T, size_t N>
static inline void T2_20_Cfg_J2V_Array(JsonVariantConst p_v, T (&p_dest)[N]) {
    if (p_v.isNull()) return;
    if (p_v.is<JsonArrayConst>()) {
        JsonArrayConst v_arr = p_v.as<JsonArrayConst>();
        size_t i = 0;
        for (JsonVariantConst v_item : v_arr) {
            if (i >= N) break;
            p_dest[i] = v_item.as<T>();
            i++;
        }
    } else {
        T v_val = p_v.as<T>();
        for (size_t i = 0; i < N; i++) p_dest[i] = v_val;
    }
}

// V2J : C 배열(float 전용) → JSON 배열
template<size_t N>
static void T2_20_Cfg_V2J_Array(JsonArray p_arr, const float (&p_src)[N]) {
    for (size_t i = 0; i < N; ++i) p_arr.add(p_src[i]);
}

// ----------------------------------------------------------------------------
// 4. DSP 설정 블록 (ST_Dsp_Config_t)
// ----------------------------------------------------------------------------

// J2V : JSON 객체 → ST_Dsp_Config_t 구조체
static void T2_20_Cfg_J2V_DspCfg(JsonObjectConst p_obj, T2_Type::ST_Dsp_Config_t& p_dsp) {
    if (p_obj.isNull()) return;
    T2_20_Cfg_J2V_Val(p_obj["rem_dc"],  p_dsp.rem_dc);
    T2_20_Cfg_J2V_Val(p_obj["med_en"],  p_dsp.med_en);
    T2_20_Cfg_J2V_Val(p_obj["med_win"], p_dsp.med_win);

    JsonObjectConst v_hpf = p_obj["hpf"];
    if (!v_hpf.isNull()) {
        T2_20_Cfg_J2V_Val(v_hpf["en"],     p_dsp.hpf.en);
        T2_20_Cfg_J2V_Val(v_hpf["cutoff"], p_dsp.hpf.cutoff);
        T2_20_Cfg_J2V_Val(v_hpf["taps"],   p_dsp.hpf.taps);
    }
    JsonObjectConst v_lpf = p_obj["lpf"];
    if (!v_lpf.isNull()) {
        T2_20_Cfg_J2V_Val(v_lpf["en"],     p_dsp.lpf.en);
        T2_20_Cfg_J2V_Val(v_lpf["cutoff"], p_dsp.lpf.cutoff);
        T2_20_Cfg_J2V_Val(v_lpf["taps"],   p_dsp.lpf.taps);
    }
    JsonObjectConst v_ihpf = p_obj["iir_hpf"];
    if (!v_ihpf.isNull()) {
        T2_20_Cfg_J2V_Val(v_ihpf["en"],     p_dsp.iir_hpf.en);
        T2_20_Cfg_J2V_Val(v_ihpf["cutoff"], p_dsp.iir_hpf.cutoff);
        T2_20_Cfg_J2V_Val(v_ihpf["q"],      p_dsp.iir_hpf.q);
    }
    JsonObjectConst v_ilpf = p_obj["iir_lpf"];
    if (!v_ilpf.isNull()) {
        T2_20_Cfg_J2V_Val(v_ilpf["en"],     p_dsp.iir_lpf.en);
        T2_20_Cfg_J2V_Val(v_ilpf["cutoff"], p_dsp.iir_lpf.cutoff);
        T2_20_Cfg_J2V_Val(v_ilpf["q"],      p_dsp.iir_lpf.q);
    }
    JsonObjectConst v_notch = p_obj["notch"];
    if (!v_notch.isNull()) {
        T2_20_Cfg_J2V_Val(v_notch["en"],   p_dsp.notch.en);
        T2_20_Cfg_J2V_Val(v_notch["freq"], p_dsp.notch.freq);
        T2_20_Cfg_J2V_Val(v_notch["gain"], p_dsp.notch.gain);
        T2_20_Cfg_J2V_Val(v_notch["q"],    p_dsp.notch.q);
    }
    JsonObjectConst v_notch2 = p_obj["notch2"];
    if (!v_notch2.isNull()) {
        T2_20_Cfg_J2V_Val(v_notch2["en"],   p_dsp.notch2.en);
        T2_20_Cfg_J2V_Val(v_notch2["freq"], p_dsp.notch2.freq);
        T2_20_Cfg_J2V_Val(v_notch2["gain"], p_dsp.notch2.gain);
        T2_20_Cfg_J2V_Val(v_notch2["q"],    p_dsp.notch2.q);
    }
    if (!p_obj["win_type"].isNull()) {
        p_dsp.win_type = (T2_Type::EM_WindowType_t)p_obj["win_type"].as<uint8_t>();
    }
}

// V2J : ST_Dsp_Config_t 구조체 → JSON 객체
static void T2_20_Cfg_V2J_DspCfg(JsonObject p_obj, const T2_Type::ST_Dsp_Config_t& p_dsp) {
    p_obj["rem_dc"]  = p_dsp.rem_dc;
    p_obj["med_en"]  = p_dsp.med_en;
    p_obj["med_win"] = p_dsp.med_win;

    JsonObject v_hpf = p_obj["hpf"].to<JsonObject>();
    v_hpf["en"]     = p_dsp.hpf.en;
    v_hpf["cutoff"] = p_dsp.hpf.cutoff;
    v_hpf["taps"]   = p_dsp.hpf.taps;

    JsonObject v_lpf = p_obj["lpf"].to<JsonObject>();
    v_lpf["en"]     = p_dsp.lpf.en;
    v_lpf["cutoff"] = p_dsp.lpf.cutoff;
    v_lpf["taps"]   = p_dsp.lpf.taps;

    JsonObject v_ihpf = p_obj["iir_hpf"].to<JsonObject>();
    v_ihpf["en"]     = p_dsp.iir_hpf.en;
    v_ihpf["cutoff"] = p_dsp.iir_hpf.cutoff;
    v_ihpf["q"]      = p_dsp.iir_hpf.q;

    JsonObject v_ilpf = p_obj["iir_lpf"].to<JsonObject>();
    v_ilpf["en"]     = p_dsp.iir_lpf.en;
    v_ilpf["cutoff"] = p_dsp.iir_lpf.cutoff;
    v_ilpf["q"]      = p_dsp.iir_lpf.q;

    JsonObject v_notch = p_obj["notch"].to<JsonObject>();
    v_notch["en"]   = p_dsp.notch.en;
    v_notch["freq"] = p_dsp.notch.freq;
    v_notch["gain"] = p_dsp.notch.gain;
    v_notch["q"]    = p_dsp.notch.q;

    JsonObject v_notch2 = p_obj["notch2"].to<JsonObject>();
    v_notch2["en"]   = p_dsp.notch2.en;
    v_notch2["freq"] = p_dsp.notch2.freq;
    v_notch2["gain"] = p_dsp.notch2.gain;
    v_notch2["q"]    = p_dsp.notch2.q;

    p_obj["win_type"] = (uint8_t)p_dsp.win_type;
}

// ----------------------------------------------------------------------------
// 5. 대역(Bands) 정보
// ----------------------------------------------------------------------------

// J2V : JSON bands 배열 → 센서별 대역 변수들
template<typename T_BandEn, typename T_BandStart, typename T_BandEnd,
         typename T_BandThresh, size_t AxisCount, size_t BandMax>
static uint8_t T2_20_Cfg_J2V_Bands(JsonArrayConst p_bands,
                                   T_BandEn& p_bandEn,
                                   T_BandStart& p_bandStart,
                                   T_BandEnd& p_bandEnd,
                                   T_BandThresh& p_bandThresh) {
    uint8_t i = 0;
    for (JsonObjectConst v_b : p_bands) {
        if (i >= BandMax) break;
        T2_20_Cfg_J2V_Val(v_b["en"],    p_bandEn[i]);
        T2_20_Cfg_J2V_Val(v_b["start"], p_bandStart[i]);
        T2_20_Cfg_J2V_Val(v_b["end"],   p_bandEnd[i]);

        if (!v_b["thresh"].isNull()) {
            if (v_b["thresh"].is<JsonArrayConst>()) {
                JsonArrayConst v_arr = v_b["thresh"].as<JsonArrayConst>();
                for (size_t a = 0; a < AxisCount; a++) {
                    if (a < v_arr.size()) p_bandThresh[a][i] = v_arr[a].as<float>();
                }
            } else {
                float v_val = v_b["thresh"].as<float>();
                for (size_t a = 0; a < AxisCount; a++) {
                    p_bandThresh[a][i] = v_val;
                }
            }
        }
        i++;
    }
    return i;
}

// V2J : 센서별 대역 변수들 → JSON bands 배열
template<typename T_BandEn, typename T_BandStart, typename T_BandEnd,
         typename T_BandThresh, size_t AxisCount>
static void T2_20_Cfg_V2J_Bands(JsonArray p_arr,
                                uint8_t p_activeCount,
                                const T_BandEn& p_bandEn,
                                const T_BandStart& p_bandStart,
                                const T_BandEnd& p_bandEnd,
                                const T_BandThresh& p_bandThresh) {
    for (uint8_t i = 0; i < p_activeCount; i++) {
        JsonObject v_b = p_arr.add<JsonObject>();
        v_b["en"]    = (bool)p_bandEn[i];
        v_b["start"] = p_bandStart[i];
        v_b["end"]   = p_bandEnd[i];

        JsonArray v_thArr = v_b["thresh"].to<JsonArray>();
        for (size_t a = 0; a < AxisCount; a++) {
            v_thArr.add(p_bandThresh[a][i]);
        }
    }
}
```

🔄 기존 코드 대체 가이드

기존 함수/코드 → 신규 함수 비고
cfgSet(p_v, p_dest) T2_20_Cfg_J2V_Val(p_v, p_dest)
cfgSet(p_v, bitset::ref) T2_20_Cfg_J2V_Val(p_v, ref)
T2_20_Cfg_StrCopy(p_v, buf, size) T2_20_Cfg_J2V_Str(p_v, buf, size)
jsonObj["key"] = someString; T2_20_Cfg_V2J_Str(obj, "key", str)
p_obj["key"] = value; T2_20_Cfg_V2J_Val(p_obj, "key", value)
T2_20_Cfg_SetArray(p_v, arr) T2_20_Cfg_J2V_Array(p_v, arr)
T2_20_Cfg_AddFloatExact(arr, val) 직접 arr.add(val) 또는 T2_20_Cfg_V2J_Array 사용 배열 전체를 한 번에 직렬화할 때
parseDspConfig(obj, dsp) T2_20_Cfg_J2V_DspCfg(obj, dsp)
serializeDspConfig(obj, dsp) T2_20_Cfg_V2J_DspCfg(obj, dsp)
bands 파싱 반복문 T2_20_Cfg_J2V_Bands<AXIS_MAX, BAND_MAX>(...) active_band_count 반환
bands 직렬화 반복문 T2_20_Cfg_V2J_Bands(arr, count, ...)

이제 _applyJson()과 save() 내에서 이 공통 함수들을 사용하면 코드 중복 없이 일관된 방식으로 JSON 처리가 가능합니다.

기존 _applyJson()과 save()에는 system, wifi, mqtt 등 DSP 외 여러 도메인에서 JSON ↔ 구조체 변환 코드가 반복되고 있습니다.
이 부분 역시 J2V / V2J 함수 쌍으로 일반화하면 코드 중복이 대폭 줄고 가독성이 향상됩니다.

✅ 일반화 방안

각 하위 설정 그룹에 대해 아래와 같은 패턴의 함수를 만듭니다.

· T2_20_Cfg_J2V_<Group>Cfg(JsonObjectConst obj, 구조체& dest)
· T2_20_Cfg_V2J_<Group>Cfg(JsonObject obj, const 구조체& src)

그룹 목록 (코드 기준):

그룹 주요 필드 비고
System site_id, tele_hz, wave_hz, op_mode, watchdog_ms
Wifi mode, ap_ssid, ap_pw, ap_ip, multi_ap[], disconnect_delay_ms
Mqtt enable, broker, port, id, pw, topic_root, lwt_topic, qos, proto_ver
Ntp ntp_server1, ntp_server2, ntp_tz
Storage rot_mb, rot_min, save_raw, keep_max, idle_flush_ms, pre_trig_sec
Output enabled, output_sequence, sequence_frames
Decision max_trial_count, sta_lta_threshold, min_trigger_count, valid_start_sec, valid_end_sec
Trigger hold_ms, use_sleep, sleep_sec _dynConfig의 직속 필드
Accel enable, axis_mask, range, odr, … + dsp, bands, 배열들 dsp, bands는 이미 공통 함수 사용
Gyro enable, axis_mask, range, … + dsp, bands, 배열들
Audio enable, channel_mask, sample_rate, … + noise, eq_coeffs, dsp, bands noise는 하위 객체

```cpp
static void T2_20_Cfg_J2V_WifiCfg(JsonObjectConst p_obj, T2_Type::ST_WiFiConfig_t& p_wifi) {
    if (p_obj.isNull()) return;
    if (!p_obj["mode"].isNull())
        p_wifi.mode = (T2_Type::EM_WiFiMode_t)p_obj["mode"].as<uint8_t>();
    T2_20_Cfg_J2V_Str(p_obj["ap_ssid"], p_wifi.ap_ssid, sizeof(p_wifi.ap_ssid));
    T2_20_Cfg_J2V_Str(p_obj["ap_pw"],   p_wifi.ap_pw,   sizeof(p_wifi.ap_pw));
    T2_20_Cfg_J2V_Str(p_obj["ap_ip"],   p_wifi.ap_ip,   sizeof(p_wifi.ap_ip));

    JsonArrayConst v_multi = p_obj["multi_ap"];
    if (!v_multi.isNull()) {
        uint8_t i = 0;
        for (JsonObjectConst v_ap : v_multi) {
            if (i >= T2_Def::Global::NetLimit::NET_MULTI_AP_MAX) break;
            T2_20_Cfg_J2V_Str(v_ap["ssid"], p_wifi.multi_ssid[i], sizeof(p_wifi.multi_ssid[i]));
            T2_20_Cfg_J2V_Str(v_ap["pw"],   p_wifi.multi_pw[i],   sizeof(p_wifi.multi_pw[i]));
            i++;
        }
    }
    T2_20_Cfg_J2V_Val(p_obj["disconnect_delay_ms"], p_wifi.disconnect_delay_ms);
}
```

```cpp
#pragma once

#include <cstring>
#include <bitset>

// ----------------------------------------------------------------------------
// 1. 기본 유틸 – 문자열
// ----------------------------------------------------------------------------

// JSON → char[]
inline void T2_20_Cfg_J2V_Str(JsonVariantConst p_v, char* p_dest, size_t p_size) {
    if (!p_v.isNull()) {
        const char* src = p_v.as<const char*>();
        if (src) strlcpy(p_dest, src, p_size);
    }
}

// char[] → JSON 객체의 키
inline void T2_20_Cfg_V2J_Str(JsonObject p_obj, const char* key, const char* src) {
    p_obj[key] = src;
}

// ----------------------------------------------------------------------------
// 2. 기본 유틸 – 스칼라 / enum / bitset
// ----------------------------------------------------------------------------

// JSON → 일반 변수
template<typename T>
inline void T2_20_Cfg_J2V_Val(JsonVariantConst p_v, T& p_dest) {
    if (!p_v.isNull()) p_dest = p_v.as<T>();
}

// bitset::reference 특수화
inline void T2_20_Cfg_J2V_Val(JsonVariantConst p_v, std::bitset<16>::reference p_dest) {
    if (!p_v.isNull()) p_dest = p_v.as<bool>();
}

// 일반 변수 → JSON 키
template<typename T>
inline void T2_20_Cfg_V2J_Val(JsonObject p_obj, const char* key, const T& val) {
    p_obj[key] = val;
}

// ----------------------------------------------------------------------------
// 3. 기본 유틸 – 1차원 고정 배열
// ----------------------------------------------------------------------------

// JSON 배열(또는 스칼라) → C 배열
template<typename T, size_t N>
inline void T2_20_Cfg_J2V_Array(JsonVariantConst p_v, T (&p_dest)[N]) {
    if (p_v.isNull()) return;
    if (p_v.is<JsonArrayConst>()) {
        JsonArrayConst arr = p_v.as<JsonArrayConst>();
        size_t i = 0;
        for (JsonVariantConst item : arr) {
            if (i >= N) break;
            p_dest[i] = item.as<T>();
            i++;
        }
    } else {
        T val = p_v.as<T>();
        for (size_t i = 0; i < N; i++) p_dest[i] = val;
    }
}

// C 배열(float) → JSON 배열
template<size_t N>
inline void T2_20_Cfg_V2J_Array(JsonArray p_arr, const float (&p_src)[N]) {
    for (size_t i = 0; i < N; ++i) p_arr.add(p_src[i]);
}

void T2_20_Cfg_J2V_DspCfg(JsonObjectConst p_obj, T2_Type::ST_Dsp_Config_t& p_dsp) {
    if (p_obj.isNull()) return;
    T2_20_Cfg_J2V_Val(p_obj["rem_dc"],  p_dsp.rem_dc);
    T2_20_Cfg_J2V_Val(p_obj["med_en"],  p_dsp.med_en);
    T2_20_Cfg_J2V_Val(p_obj["med_win"], p_dsp.med_win);

    JsonObjectConst hpf = p_obj["hpf"];
    if (!hpf.isNull()) {
        T2_20_Cfg_J2V_Val(hpf["en"],     p_dsp.hpf.en);
        T2_20_Cfg_J2V_Val(hpf["cutoff"], p_dsp.hpf.cutoff);
        T2_20_Cfg_J2V_Val(hpf["taps"],   p_dsp.hpf.taps);
    }
    JsonObjectConst lpf = p_obj["lpf"];
    if (!lpf.isNull()) {
        T2_20_Cfg_J2V_Val(lpf["en"],     p_dsp.lpf.en);
        T2_20_Cfg_J2V_Val(lpf["cutoff"], p_dsp.lpf.cutoff);
        T2_20_Cfg_J2V_Val(lpf["taps"],   p_dsp.lpf.taps);
    }
    JsonObjectConst ihpf = p_obj["iir_hpf"];
    if (!ihpf.isNull()) {
        T2_20_Cfg_J2V_Val(ihpf["en"],     p_dsp.iir_hpf.en);
        T2_20_Cfg_J2V_Val(ihpf["cutoff"], p_dsp.iir_hpf.cutoff);
        T2_20_Cfg_J2V_Val(ihpf["q"],      p_dsp.iir_hpf.q);
    }
    JsonObjectConst ilpf = p_obj["iir_lpf"];
    if (!ilpf.isNull()) {
        T2_20_Cfg_J2V_Val(ilpf["en"],     p_dsp.iir_lpf.en);
        T2_20_Cfg_J2V_Val(ilpf["cutoff"], p_dsp.iir_lpf.cutoff);
        T2_20_Cfg_J2V_Val(ilpf["q"],      p_dsp.iir_lpf.q);
    }
    JsonObjectConst notch = p_obj["notch"];
    if (!notch.isNull()) {
        T2_20_Cfg_J2V_Val(notch["en"],   p_dsp.notch.en);
        T2_20_Cfg_J2V_Val(notch["freq"], p_dsp.notch.freq);
        T2_20_Cfg_J2V_Val(notch["gain"], p_dsp.notch.gain);
        T2_20_Cfg_J2V_Val(notch["q"],    p_dsp.notch.q);
    }
    JsonObjectConst notch2 = p_obj["notch2"];
    if (!notch2.isNull()) {
        T2_20_Cfg_J2V_Val(notch2["en"],   p_dsp.notch2.en);
        T2_20_Cfg_J2V_Val(notch2["freq"], p_dsp.notch2.freq);
        T2_20_Cfg_J2V_Val(notch2["gain"], p_dsp.notch2.gain);
        T2_20_Cfg_J2V_Val(notch2["q"],    p_dsp.notch2.q);
    }
    if (!p_obj["win_type"].isNull())
        p_dsp.win_type = (T2_Type::EM_WindowType_t)p_obj["win_type"].as<uint8_t>();
}

// JSON → bands 변수들, 반환값: 실제 읽은 밴드 수
template<typename T_BandEn, typename T_BandStart, typename T_BandEnd,
         typename T_BandThresh, size_t AxisCount, size_t BandMax>
uint8_t T2_20_Cfg_J2V_Bands(JsonArrayConst p_bands,
                            T_BandEn& p_bandEn,
                            T_BandStart& p_bandStart,
                            T_BandEnd& p_bandEnd,
                            T_BandThresh& p_bandThresh) {
    uint8_t i = 0;
    for (JsonObjectConst b : p_bands) {
        if (i >= BandMax) break;
        T2_20_Cfg_J2V_Val(b["en"],    p_bandEn[i]);
        T2_20_Cfg_J2V_Val(b["start"], p_bandStart[i]);
        T2_20_Cfg_J2V_Val(b["end"],   p_bandEnd[i]);

        if (!b["thresh"].isNull()) {
            if (b["thresh"].is<JsonArrayConst>()) {
                JsonArrayConst arr = b["thresh"].as<JsonArrayConst>();
                for (size_t a = 0; a < AxisCount; a++) {
                    if (a < arr.size()) p_bandThresh[a][i] = arr[a].as<float>();
                }
            } else {
                float v = b["thresh"].as<float>();
                for (size_t a = 0; a < AxisCount; a++) p_bandThresh[a][i] = v;
            }
        }
        i++;
    }
    return i;
}

// bands 변수들 → JSON 배열
template<typename T_BandEn, typename T_BandStart, typename T_BandEnd,
         typename T_BandThresh, size_t AxisCount>
void T2_20_Cfg_V2J_Bands(JsonArray p_arr,
                         uint8_t p_activeCount,
                         const T_BandEn& p_bandEn,
                         const T_BandStart& p_bandStart,
                         const T_BandEnd& p_bandEnd,
                         const T_BandThresh& p_bandThresh) {
    for (uint8_t i = 0; i < p_activeCount; i++) {
        JsonObject b = p_arr.add<JsonObject>();
        b["en"]    = (bool)p_bandEn[i];
        b["start"] = p_bandStart[i];
        b["end"]   = p_bandEnd[i];

        JsonArray th = b["thresh"].to<JsonArray>();
        for (size_t a = 0; a < AxisCount; a++) th.add(p_bandThresh[a][i]);
    }
}

// 6.2 Wifi
inline void T2_20_Cfg_J2V_WifiCfg(JsonObjectConst obj, T2_Type::ST_WiFiConfig_t& wifi) {
    if (obj.isNull()) return;
    if (!obj["mode"].isNull())
        wifi.mode = (T2_Type::EM_WiFiMode_t)obj["mode"].as<uint8_t>();
    T2_20_Cfg_J2V_Str(obj["ap_ssid"], wifi.ap_ssid, sizeof(wifi.ap_ssid));
    T2_20_Cfg_J2V_Str(obj["ap_pw"],   wifi.ap_pw,   sizeof(wifi.ap_pw));
    T2_20_Cfg_J2V_Str(obj["ap_ip"],   wifi.ap_ip,   sizeof(wifi.ap_ip));

    JsonArrayConst multi = obj["multi_ap"];
    if (!multi.isNull()) {
        uint8_t i = 0;
        for (JsonObjectConst ap : multi) {
            if (i >= T2_Def::Global::NetLimit::NET_MULTI_AP_MAX) break;
            T2_20_Cfg_J2V_Str(ap["ssid"], wifi.multi_ssid[i], sizeof(wifi.multi_ssid[i]));
            T2_20_Cfg_J2V_Str(ap["pw"],   wifi.multi_pw[i],   sizeof(wifi.multi_pw[i]));
            i++;
        }
    }
    T2_20_Cfg_J2V_Val(obj["disconnect_delay_ms"], wifi.disconnect_delay_ms);
}

inline void T2_20_Cfg_V2J_WifiCfg(JsonObject obj, const T2_Type::ST_WiFiConfig_t& wifi) {
    T2_20_Cfg_V2J_Val(obj, "mode", (uint8_t)wifi.mode);
    T2_20_Cfg_V2J_Str(obj, "ap_ssid", wifi.ap_ssid);
    T2_20_Cfg_V2J_Str(obj, "ap_pw",   wifi.ap_pw);
    T2_20_Cfg_V2J_Str(obj, "ap_ip",   wifi.ap_ip);

    JsonArray multi = obj["multi_ap"].to<JsonArray>();
    for (uint8_t i = 0; i < T2_Def::Global::NetLimit::NET_MULTI_AP_MAX; i++) {
        JsonObject ap = multi.add<JsonObject>();
        T2_20_Cfg_V2J_Str(ap, "ssid", wifi.multi_ssid[i]);
        T2_20_Cfg_V2J_Str(ap, "pw",   wifi.multi_pw[i]);
    }
    T2_20_Cfg_V2J_Val(obj, "disconnect_delay_ms", wifi.disconnect_delay_ms);
}

// 7.1 Accel
inline void T2_20_Cfg_J2V_AccelCfg(JsonObjectConst obj, T2_Type::ST_AccelConfig_t& acc) {
    if (obj.isNull()) return;

    T2_20_Cfg_J2V_Val(obj["enable"],         acc.enable);
    T2_20_Cfg_J2V_Val(obj["axis_mask"],      acc.axis_mask);
    T2_20_Cfg_J2V_Val(obj["range"],          acc.range);
    T2_20_Cfg_J2V_Val(obj["odr"],            acc.odr);
    T2_20_Cfg_J2V_Val(obj["bwp"],            acc.bwp);
    T2_20_Cfg_J2V_Val(obj["filter_perf"],    acc.filter_perf);
    T2_20_Cfg_J2V_Val(obj["fifo_watermark"], acc.fifo_watermark);
    T2_20_Cfg_J2V_Val(obj["sample_rate"],    acc.sample_rate);
    T2_20_Cfg_J2V_Val(obj["fft_size"],       acc.fft_size);
    T2_20_Cfg_J2V_Val(obj["noise_perf"],     acc.noise_perf);

    T2_20_Cfg_J2V_Val(obj["motion_en"], acc.motion_en);
```
